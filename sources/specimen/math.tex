\parttitle{01}{Complex analysis}
{\bfseries Residue-theorem specimen.}
Suppose $f$ is holomorphic throughout $G$ apart from finitely many isolated
singularities $a_1,a_2,\dots,a_m$. A rectifiable closed path $\gamma$ avoids
these points and is null-homologous, $\gamma\approx0$, in $G$. The test deliberately nests a bold
letter and a double-struck space inside the exponent:
\[
  \frac{1}{2\pi i}\int\limits_\gamma
  f\Bigl(x^{\mathbf{N}\in\mathbb{C}^{N\times10}}\Bigr)
  =\sum_{k=1}^{m}n(\gamma;a_k)\Res(f;a_k)\,.
\]
{\bfseries Maximum modulus.}
On a bounded domain $G\subset\mathbb{C}$, let $f$ extend continuously to the
closure $G^-$ and be holomorphic inside. Its maximum occurs on the boundary:
\[
  \max\{\,|f(z)|:z\in G^-\,\}
  =\max\{\,|f(z)|:z\in\partial G\,\}\,.
\]

\parttitle{02}{Large operators, in text and on display}
At text size,
$\iiint\limits_Q f(x,y,z)\diff x\diff y\diff z$
and $\prod_{\gamma\in\Gamma_{\bar C}}\partial(\tilde X_\gamma)$
stay part of the sentence. A display brings out the larger drawings:
\[
  \iiiint\limits_Q f(w,x,y,z)\diff w\diff x\diff y\diff z
  \leq\oint_{\partial Q}f'\Biggl(\max\Biggl\{
  \frac{\Vert w\Vert}{\vert w^2+x^2\vert};\quad
  \frac{\Vert z\Vert}{\vert y^2+z^2\vert};\quad
  \frac{\Vert w\oplus z\Vert}{\vert x\oplus y\vert}
  \Biggr\}\Biggr)\,.
\]
